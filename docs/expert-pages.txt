// --- 通用 UI 组件 (模拟 Ant Design 风格定制) ---
const Button = ({ children, type = 'default', danger = false, disabled = false, className = '', onClick }) => {
  let baseStyle = "px-4 py-2 rounded-md font-medium transition-colors inline-flex items-center justify-center gap-2 text-sm ";
  if (disabled) {
    baseStyle += "bg-gray-100 text-gray-400 border border-gray-200 cursor-not-allowed ";
  } else if (type === 'primary' && danger) {
    baseStyle += "bg-[#bf3030] text-white hover:bg-red-800 border border-transparent ";
  } else if (type === 'primary') {
    baseStyle += "bg-[#145c72] text-white hover:bg-[#104b5d] border border-transparent ";
  } else if (danger) {
    baseStyle += "bg-white text-[#bf3030] hover:bg-red-50 border border-[#bf3030] ";
  } else if (type === 'text') {
    baseStyle += "bg-transparent text-[#132235] hover:bg-gray-100 border-transparent px-2 ";
  } else {
    baseStyle += "bg-white text-[#132235] hover:bg-gray-50 border border-[#d8e2ea] ";
  }
  return <button className={`${baseStyle} ${className}`} disabled={disabled} onClick={onClick}>{children}</button>;
};

const Badge = ({ status, text, color, dot = false }) => {
  const colors = {
    processing: "bg-[#1d6fd8] text-white border-transparent",
    warning: "bg-orange-100 text-orange-800 border-orange-200",
    error: "bg-red-100 text-red-800 border-red-200",
    success: "bg-green-100 text-green-800 border-green-200"
  };

  if (dot) {
    return (
      <span className="inline-flex items-center gap-1.5 text-sm font-medium text-red-600">
        <span className="relative flex h-2 w-2">
          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-red-400 opacity-75"></span>
          <span className="relative inline-flex rounded-full h-2 w-2 bg-red-500"></span>
        </span>
        {text}
      </span>
    );
  }

  return (
    <span className={`inline-flex items-center px-2.5 py-0.5 rounded border text-xs font-medium ${color || colors[status]}`}>
      {text}
    </span>
  );
};

const Card = ({ children, className = '' }) => (

);

const Modal = ({ isOpen, onClose, onConfirm, title, children, confirmDisabled }) => {
  if (!isOpen) return null;
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 backdrop-blur-sm animate-in fade-in duration-200">
      <div className="bg-white rounded-xl shadow-xl w-full max-w-md overflow-hidden animate-in zoom-in-95 duration-200">
        <div className="px-6 py-4 border-b border-gray-100 flex justify-between items-center bg-[#f7fafc]">
          <h3 className="font-semibold text-lg text-[#132235] flex items-center gap-2">
            <ShieldAlert className="w-5 h-5 text-[#b76504]" />
            {title}
          </h3>
          <button onClick={onClose} className="text-gray-400 hover:text-gray-600"><XSquare className="w-5 h-5"/></button>
        </div>
        <div className="p-6">
          {children}
        </div>
        <div className="px-6 py-4 bg-gray-50 border-t border-gray-100 flex justify-end gap-3">
          <Button onClick={onClose}>取消</Button>
          <Button type="primary" onClick={onConfirm} disabled={confirmDisabled} className="bg-[#145c72]">确认发布</Button>
        </div>
      </div>
    </div>

};
