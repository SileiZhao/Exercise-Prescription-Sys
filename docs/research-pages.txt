// --- 通用 UI 组件 (模拟 Ant Design 风格定制) ---
const Button = ({ children, type = 'default', danger = false, disabled = false, className = '', onClick }) => {
  let baseStyle = "px-4 py-2 rounded-md font-medium transition-colors inline-flex items-center justify-center gap-2 text-sm ";
  if (disabled) {
    baseStyle += "bg-gray-100 text-gray-400 border border-gray-200 cursor-not-allowed ";
  } else if (type === 'primary' && danger) {
    baseStyle += "bg-[#bf3030] text-white hover:bg-red-800 border border-transparent ";
  } else if (type === 'primary') {
    baseStyle += "bg-[#145c72] text-white hover:bg-[#104b5d] border border-transparent ";
  } else if (danger) {
    baseStyle += "bg-white text-[#bf3030] hover:bg-red-50 border border-[#bf3030] ";
  } else if (type === 'text') {
    baseStyle += "bg-transparent text-[#132235] hover:bg-gray-100 border-transparent px-2 ";
  } else {
    baseStyle += "bg-white text-[#132235] hover:bg-gray-50 border border-[#d8e2ea] ";
  }
  return <button className={`${baseStyle} ${className}`} disabled={disabled} onClick={onClick}>{children}</button>;
};

const Badge = ({ status, text, color }) => {
  const colors = {
    processing: "bg-blue-100 text-blue-800 border-blue-200",
    warning: "bg-orange-100 text-orange-800 border-orange-200",
    error: "bg-red-100 text-red-800 border-red-200",
    success: "bg-[#f0f9f4] text-[#228b4e] border-[#228b4e]/30"
  };

  return (
    <span className={`inline-flex items-center px-2.5 py-0.5 rounded border text-xs font-medium ${color || colors[status] || 'bg-gray-100 text-gray-800 border-gray-200'}`}>
      {text}
    </span>
  );
};

const Card = ({ title, children, className = '', bodyClass = '' }) => (
  <div className={`bg-white rounded-lg border border-[#d8e2ea] shadow-sm overflow-hidden ${className}`}>
    {title && (
      <div className="px-5 py-3 border-b border-[#d8e2ea] bg-white font-semibold text-[#132235] flex items-center gap-2">
        {title}
      </div>
    )}
    <div className={`p-5 ${bodyClass}`}>
      {children}
    </div>
  </div>
);

const Alert = ({ type = 'info', message, description, icon: Icon, className = '' }) => {
  const styles = {
    info: "bg-blue-50 border-blue-200 text-blue-800",
    success: "bg-[#f0f9f4] border-[#228b4e]/30 text-[#228b4e]",
    warning: "bg-yellow-50 border-yellow-200 text-yellow-800",
    error: "bg-red-50 border-red-200 text-red-800"
  };

  return (
    <div className={`p-4 rounded-md border flex gap-3 ${styles[type]} ${className}`}>
      {Icon && <Icon className="w-5 h-5 shrink-0 mt-0.5" />}
      <div>
        <div className="font-medium text-sm">{message}</div>
        {description && <div className="text-sm mt-1 opacity-90">{description}</div>}
      </div>
    </div>
  );
};
